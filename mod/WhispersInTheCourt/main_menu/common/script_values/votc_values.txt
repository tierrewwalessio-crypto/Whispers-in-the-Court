# Whispers in the Court - balance envelope.
#
# ONE source of truth for every magnitude the AI is allowed to move.
# The middleware (Court Brain) mirrors these numbers, but the game clamps
# again here, so a hallucinating model can never exceed the envelope.
#
# All the *_weak_* / *_mild_* / *_severe_* constants used by the action
# catalogue come from vanilla (main_menu/common/script_values/default_values.txt)
# so a VotC outcome is always the size of an ordinary vanilla event outcome.

# and is capped, so a long AI session cannot snowball.

# ------------------------------------------------------------------ gold band
# Gold moved by a scene - given, spent, paid, taken - is never a flat number.
# It follows three things, so that it stays in proportion for every realm:
#  - its economy: a fraction of the tax base (read as vanilla reads it in
#    common/country_interactions/ask_for_money.txt);
#  - what its chest already holds: a small share of the treasury on top - a rich
#    Crown gives and spends more, an empty one does not (never below nothing);
#  - what the realms of its rank hold: never more than a share of their usual
#    treasury (var:votc_peer_gold, the median of its peers, written by Court Brain
#    from the latest save), so a hoard does not make sums absurd for the world.
votc_gold_weak = {
	value = country_tax_base
	multiply = 0.25
	add = { value = gold multiply = 0.02 min = 0 }
	if = {
		limit = { has_variable = votc_peer_gold }
		max = { value = var:votc_peer_gold multiply = 0.15 min = 5 }
	}
	min = 1
	max = 300
	round = yes
}
votc_gold_mild = {
	value = country_tax_base
	multiply = 0.75
	add = { value = gold multiply = 0.05 min = 0 }
	if = {
		limit = { has_variable = votc_peer_gold }
		max = { value = var:votc_peer_gold multiply = 0.4 min = 15 }
	}
	min = 3
	max = 900
	round = yes
}
votc_gold_severe = {
	value = country_tax_base
	multiply = 2
	add = { value = gold multiply = 0.12 min = 0 }
	if = {
		limit = { has_variable = votc_peer_gold }
		max = { value = var:votc_peer_gold multiply = 1 min = 40 }
	}
	min = 10
	max = 2500
	round = yes
}

# ---------------------------------------------------------------- manpower band
# The game counts manpower in THOUSANDS (vanilla gives add_manpower = 0.1 or half a
# month's recruits). A band is some months of the realm's own recruits, with small
# floors: the minimums of 10 / 50 / 150 here once meant 10,000 - 150,000 men, far
# beyond a small realm's reserve. The game never lets it pass the realm's maximum.
votc_manpower_weak = {
	value = monthly_manpower
	multiply = 2
	min = 0.05
	max = 5
}
votc_manpower_mild = {
	value = monthly_manpower
	multiply = 6
	min = 0.2
	max = 15
}
votc_manpower_severe = {
	value = monthly_manpower
	multiply = 15
	min = 0.5
	max = 40
}
