namespace = votc

# Whispers in the Court - the scenes that arrive in game.
#
# The conversation itself happens in Court Brain's side panel. What comes
# back into the game is the RESULT: an outcome, a page of chronicle, someone
# asking to be received. Titles and bodies point at keys in
# main_menu/localization/english/votc_dynamic_l_english.yml, which Court
# Brain rewrites and the bridge widget reloads just before the event fires.
#
# Every event that tells a story offers "Rispondi di persona": it raises a
# signal and Court Brain's panel slides open on that very scene.

# =====================================================================
# 100 - The outcome of an audience, a council, an envoy, a progress
# =====================================================================

votc.100 = {
	type = country_event
	title = {
		first_valid = {
			triggered_desc = { trigger = { NOT = { votc_scene_ok = { kind = out } } } desc = votc_void_title }
			desc = votc_out_title
		}
	}
	desc = {
		first_valid = {
			triggered_desc = { trigger = { NOT = { votc_scene_ok = { kind = out } } } desc = votc_void_body }
			desc = votc_out_body
		}
	}
	outcome = neutral

	immediate = {
		var:votc_tchar ?= { save_scope_as = votc_speaker }
	}

	# The consequences are applied HERE, not before the event: so the game's
	# own tooltip on the button lists exactly what changes. The custom
	# tooltip below it says what Court Brain asked for, to compare.
	option = {
		name = votc_out_ok
		if = { limit = { votc_scene_ok = { kind = out } } votc_apply_queue = yes }
		custom_tooltip = votc_out_requested
	}
	option = {
		name = votc_opt_reply
		if = { limit = { votc_scene_ok = { kind = out } } votc_apply_queue = yes }
		custom_tooltip = votc_out_requested
		votc_signal_reply = yes
	}
	# The page was written by the royal biographer, a person of the court:
	# the ruler may go and speak with them - praise, correct, or dismiss them.
	option = {
		name = votc_opt_biographer
		if = { limit = { votc_scene_ok = { kind = out } } votc_apply_queue = yes }
		custom_tooltip = votc_out_requested
		votc_signal_biographer = yes
	}
	option = {
		name = votc_opt_book
		if = { limit = { votc_scene_ok = { kind = out } } votc_apply_queue = yes }
		custom_tooltip = votc_out_requested
		custom_tooltip = votc_book_tt
		votc_signal_book = yes
	}
}

# =====================================================================
# 500 - A story: a situation with choices, often one stage of a longer arc
# =====================================================================
#
# Court Brain writes the situation and three choices; each choice has its own
# consequences, parked in its own slots (votc_qo<n>_*), so the tooltip of
# every button shows exactly what that choice does. The choice is reported
# back (votc_signal_arc) and the story may go on in a later event.

votc.500 = {
	type = country_event
	title = {
		first_valid = {
			triggered_desc = { trigger = { NOT = { votc_scene_ok = { kind = arc } } } desc = votc_void_title }
			desc = votc_arc_title
		}
	}
	desc = {
		first_valid = {
			triggered_desc = { trigger = { NOT = { votc_scene_ok = { kind = arc } } } desc = votc_void_body }
			desc = votc_arc_body
		}
	}
	outcome = neutral

	option = {
		name = votc_arc_opt1
		trigger = { votc_scene_ok = { kind = arc } }
		votc_apply_option = { o = 1 }
		custom_tooltip = votc_arc_req1
		votc_signal_arc = { choice = 1 }
	}
	option = {
		name = votc_arc_opt2
		trigger = { votc_scene_ok = { kind = arc } }
		votc_apply_option = { o = 2 }
		custom_tooltip = votc_arc_req2
		votc_signal_arc = { choice = 2 }
	}
	option = {
		name = votc_arc_opt3
		trigger = { votc_scene_ok = { kind = arc } }
		votc_apply_option = { o = 3 }
		custom_tooltip = votc_arc_req3
		votc_signal_arc = { choice = 3 }
	}
	option = {
		name = votc_arc_reply
		trigger = { votc_scene_ok = { kind = arc } }
		votc_queue_clear_options = yes
		votc_signal_arc = { choice = 9 }
	}
	option = {
		name = votc_void_ok
		trigger = { NOT = { votc_scene_ok = { kind = arc } } }
		votc_queue_clear_options = yes
	}
}
