# Whispers in the Court - core.
#
# HOW THE GAME TALKS TO COURT BRAIN (all verified in EU5 1.3.11, release build)
#
#   game -> middleware
#     A script effect raises a SIGNAL variable on the player country.
#     The invisible widget in gui/votc_bridge.gui watches those variables
#     (state trigger_when). When one goes up it runs, through
#     ExecuteConsoleCommandsForced, a list of "votc_say ..." commands built
#     from localisation templates full of data functions. votc_say is not a
#     real command: it fails harmlessly, but the console writes every command
#     it receives to Documents/.../console_history.txt, which Court Brain
#     tails. The last command of the list lowers the signal again.
#
#   middleware -> game
#     Court Brain writes run/votc_poll.txt and run/votc_in.txt. The bridge
#     widget runs votc_poll.txt every few seconds; when it announces new mail
#     (votc_mail), the widget reloads localisation if needed and runs
#     votc_in.txt, which applies the consequences and fires the events.
#
# The script-side logging effects (debug_log, error_log) documented in the
# executable do NOT exist in the release build, which is why none are used.
#
# Scope convention: country scope, the human player, unless stated.

# =====================================================================
# Bookkeeping
# =====================================================================

votc_init_vars = {
	if = { limit = { NOT = { has_variable = votc_budget } } set_variable = { name = votc_budget value = votc_budget_start } }
	if = { limit = { NOT = { has_variable = votc_mail_seq } } set_variable = { name = votc_mail_seq value = 0 } }
	if = { limit = { NOT = { has_variable = votc_scene_actions } } set_variable = { name = votc_scene_actions value = 0 } }
	if = { limit = { NOT = { has_variable = votc_scene_spent } } set_variable = { name = votc_scene_spent value = 0 } }
	if = { limit = { NOT = { has_variable = votc_spend_ok } } set_variable = { name = votc_spend_ok value = 0 } }
	if = { limit = { NOT = { has_variable = votc_refusals } } set_variable = { name = votc_refusals value = 0 } }
	if = { limit = { NOT = { has_variable = votc_agenda } } set_variable = { name = votc_agenda value = 0 } }
}

# =====================================================================
# Context preparation
# =====================================================================
#
# The localisation templates can only follow variables, not iterate lists,
# so the people the model should know about are parked in variables first.
# Empty slots are filled with the ruler; Court Brain drops duplicates.

votc_prepare_context = {
	save_scope_as = votc_country
	votc_init_vars = yes
	votc_regen_budget = yes

	# EU5 does not let an effect read back a variable it set a moment ago, so
	# nothing below reads what it writes: each slot is filled directly from
	# the nth best person (vanilla does the same with position = N in
	# common/scripted_effects/situation_effects.txt).
	if = {
		limit = { has_ruler = yes }
		ruler ?= { scope:votc_country = { set_variable = { name = votc_ruler value = prev } } }
	}
	if = {
		limit = { has_heir = yes }
		heir ?= { scope:votc_country = { set_variable = { name = votc_heir value = prev } } }
	}
	else_if = {
		limit = { has_ruler = yes }
		ruler ?= { scope:votc_country = { set_variable = { name = votc_heir value = prev } } }
	}

	# Clear the six slots to the ruler first; real people overwrite them.
	if = {
		limit = { has_ruler = yes }

			scope:votc_country = {
				set_variable = { name = votc_p1 value = prev }
				set_variable = { name = votc_p2 value = prev }
				set_variable = { name = votc_p3 value = prev }

	}
	ordered_character = {

			owner = scope:votc_country

				is_adult = yes
				is_ruler = no

# Talk to a character. $who$ = the character.
votc_signal_talk = {
	set_variable = { name = votc_tchar value = $who$ }
	votc_prepare_context = yes
	set_variable = { name = votc_awaiting value = 1 }
	votc_poll_fast_on = yes
	set_variable = { name = votc_sig_talk value = 1 }
}

# 2 meeting in person, 3 state visit (carried to Court Brain as the agenda).

	set_variable = { name = votc_agenda value = $mode$ }

# The bridge polls for mail every 3 seconds while this is raised, and every 12 seconds
# otherwise: in debug mode every console command costs the game a moment, and most of
# the time nothing is on its way. Raised by every signal of the player (an answer is
# coming) and by Court Brain whenever it has more to send; lowered by Court Brain's
# last mail when it has nothing more to say.
votc_poll_fast_on = {
	set_variable = { name = votc_poll_fast value = 1 }
}
votc_poll_fast_off = {
	remove_variable = votc_poll_fast
}

# =====================================================================
# Mail from the middleware
# =====================================================================

# Written into run/votc_poll.txt by Court Brain. The poll runs every few
# seconds; the sequence guard makes a repeated poll of the same mail a no-op.
# Two variants because argument substitution is textual ("1 = 1" is not a
# trigger): votc_mail_loc also asks for a localisation reload, which is how
# new AI-written text reaches the events.
votc_mail = {
	votc_init_vars = yes
	if = {
		limit = { var:votc_mail_seq < $seq$ }
		set_variable = { name = votc_mail_seq value = $seq$ }
		set_variable = { name = votc_has_mail value = 1 }
	}
}

votc_mail_loc = {
	votc_init_vars = yes
	if = {
		limit = { var:votc_mail_seq < $seq$ }
		set_variable = { name = votc_mail_seq value = $seq$ }
		votc_request_reload = yes
	}
}

votc_mail_save = {
	votc_init_vars = yes
	if = {
		limit = { var:votc_mail_seq < $seq$ }
		set_variable = { name = votc_mail_seq value = $seq$ }
		set_variable = { name = votc_has_mail_save value = 1 }
	}
}

# First line of run/votc_in.txt.
votc_in_open = {
	votc_init_vars = yes
	remove_variable = votc_has_mail
	remove_variable = votc_has_mail_loc
	remove_variable = votc_has_mail_loc_b
	remove_variable = votc_has_mail_save
	set_variable = { name = votc_scene_actions value = 0 }
	set_variable = { name = votc_scene_spent value = 0 }
}

# The conversation is over; the bridge can poll slowly again.
votc_in_close = {
	remove_variable = votc_awaiting
}

# Regenerates by ELAPSED GAME MONTHS, not by how often it is called.
votc_regen_budget = {
	# First ever call: only record the month. Reading the variables back is
	# left to the next call, because EU5 will not show them to this one.
	if = {
		limit = { NOT = { has_variable = votc_last_regen } }
		set_variable = { name = votc_last_regen value = votc_abs_month }
	}
	else_if = {
		limit = {
			has_variable = votc_budget
			var:votc_last_regen < votc_abs_month
		}
		set_variable = {
			name = votc_budget
			value = {
				value = votc_abs_month
				subtract = var:votc_last_regen
				multiply = votc_budget_monthly_regen
				add = var:votc_budget
				max = votc_budget_max
			}
		}
		set_variable = { name = votc_last_regen value = votc_abs_month }
	}
}

# =====================================================================
# Which campaign, and where its memory stands
# =====================================================================
#
# Two GLOBAL variables, so they travel inside every save file:
#   votc_campaign  a number Court Brain picks the first time it sees a game
#   votc_memhead   the last entry of Court Brain's memory journal this game
#                  state has seen
# Load an older save and the game reports the older head: Court Brain then
# remembers exactly what had happened up to that save and nothing after it.
# Written by the middleware at the top of every mail.

votc_set_campaign = {
	# Only ever on a game that has none: a mail written for one game must
	# never relabel another the player has just loaded.
	if = {
		limit = { NOT = { has_global_variable = votc_campaign } }
		set_global_variable = { name = votc_campaign value = $id$ }
		set_global_variable = { name = votc_memhead value = $head$ }
	}
}

votc_set_memhead = {
	# Moves the head forward only if this is still the game, and the point
	# of the journal, the mail was written for. If the player loaded a save
	# in the meantime the numbers differ and nothing is touched.
	if = {
		limit = {
			has_global_variable = votc_campaign
			has_global_variable = votc_memhead
			global_var:votc_campaign = $campaign$
			global_var:votc_memhead = $from$
		}
		set_global_variable = { name = votc_memhead value = $head$ }
	}
}

# =====================================================================
# Mail, version 3: the mail IS the poll file
# =====================================================================
#
# run/votc_poll.txt holds
#     if = { limit = { votc_mail_is_new = { seq = N } } votc_mail_open = { seq = N } ... }
# so each mail runs exactly once, inside the poll's own console command.

votc_mail_open = {
	votc_init_vars = yes
	set_variable = { name = votc_mail_seq value = $seq$ }
	set_variable = { name = votc_scene_actions value = 0 }
	set_variable = { name = votc_scene_spent value = 0 }
}

# The mail carries new text: the bridge reloads the localisation, then calls
# votc_loc_done, whose acknowledgement tells Court Brain the text is in.
# The bridge fires only when a flag goes UP. A reload the console once refused
# used to leave the flag up - saved with the game - and no text ever loaded
# again in that campaign. So every request raises the OTHER of two flags: one
# stuck flag can never block the next reload.
votc_request_reload = {
	if = {
		limit = { has_variable = votc_has_mail_loc }
		remove_variable = votc_has_mail_loc
		set_variable = { name = votc_has_mail_loc_b value = 1 }
	}
	else = {
		remove_variable = votc_has_mail_loc_b
		set_variable = { name = votc_has_mail_loc value = 1 }
	}
}

votc_loc_done = {
	remove_variable = votc_has_mail_loc
	remove_variable = votc_has_mail_loc_b
	set_variable = { name = votc_sig_ack value = 1 }
}

# A game just loaded: whatever was left raised by an earlier session is lowered,
# so that the first mail's flags go up cleanly.
votc_bridge_reset = {
	votc_poll_fast_on = yes           # a game just loaded: Court Brain has things to say
	remove_variable = votc_has_mail_loc
	remove_variable = votc_has_mail_loc_b
	remove_variable = votc_has_mail_loc_stuck
	remove_variable = votc_has_mail_loc_retry
	remove_variable = votc_has_mail_save
	remove_variable = votc_has_mail_logs
	remove_variable = votc_sig_ack
}

votc_request_save = {
	set_variable = { name = votc_has_mail_save value = 1 }
}

# Court Brain asks the game to empty its own log files. In debug mode every
# console command makes the engine re-check the whole script and write it all
# to debug.log (about 15 MB a minute); the bridge runs Log.ClearAll, the game's
# own command, which empties the logs safely while they are open.
votc_request_logs = {
	set_variable = { name = votc_has_mail_logs value = 1 }
}

# =====================================================================
# Watchdog - run by every poll that carries no new mail
# =====================================================================
#
# A bridge command the console refused leaves its variable raised, and a
# GUI state only fires when a variable goes UP. So a variable still raised
# two polls later is lowered, and raised again on the next poll: the state
# fires once more. (Each step reads only what earlier polls wrote.)

votc_bridge_watchdog = {
	if = { limit = { has_variable = votc_sig_arc_retry } remove_variable = votc_sig_arc_retry set_variable = { name = votc_sig_arc value = 1 } }
	else_if = { limit = { has_variable = votc_sig_arc has_variable = votc_sig_arc_stuck } remove_variable = votc_sig_arc remove_variable = votc_sig_arc_stuck set_variable = { name = votc_sig_arc_retry value = 1 } }
	else_if = { limit = { has_variable = votc_sig_arc } set_variable = { name = votc_sig_arc_stuck value = 1 } }
	else = { remove_variable = votc_sig_arc_stuck }
	if = { limit = { has_variable = votc_has_mail_loc_retry } remove_variable = votc_has_mail_loc_retry set_variable = { name = votc_has_mail_loc value = 1 } }
	else_if = { limit = { has_variable = votc_has_mail_loc has_variable = votc_has_mail_loc_stuck } remove_variable = votc_has_mail_loc remove_variable = votc_has_mail_loc_stuck set_variable = { name = votc_has_mail_loc_retry value = 1 } }
	else_if = { limit = { has_variable = votc_has_mail_loc } set_variable = { name = votc_has_mail_loc_stuck value = 1 } }
