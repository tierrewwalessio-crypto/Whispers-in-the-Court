namespace = votc

# Voices of the Court - the surface the player actually sees.
#
# Every one of these events is a shell. The title, the body and the option
# labels all point at keys in
#   main_menu/localization/english/votc_dynamic_l_english.yml
# which Court Brain rewrites and hot-reloads (console `reload localization`)
# immediately before firing the event. So the text is different every time
# while the event definition never changes.
#
# Each scene owns its own block of keys (votc_dlg_*, votc_cnc_*, ...), which
# is why a chronicle arriving mid-conversation cannot overwrite the words the
# chancellor is saying.
#
# var:votc_<scene>_nopt is how many option slots the middleware filled. The
# last option of every event is a fallback, so a scene is never a dead end
# even if the middleware dies between writing the text and firing the event.
#
# votc_execute_staged = { slot = N } is how a heavy outcome - a war, a peace,
# an alliance, a claim - actually happens. The middleware stages it against
# an option slot; clicking that option fires it. The fallback option clears
# the staging instead, so walking away from a scene never commits the realm
# to anything.

# =====================================================================
# 100 - A conversation
# =====================================================================

votc.100 = {
	type = country_event
	title = votc_dlg_title
	desc = votc_dlg_body
	outcome = neutral
	orphan = yes

	immediate = {
		votc_restore_scopes = yes
	}

	option = {
		name = votc_dlg_o1
		trigger = { var:votc_dlg_nopt >= 1 }
		votc_execute_staged = { slot = 1 }
		votc_talk_reply = { choice = 1 }
	}
	option = {
		name = votc_dlg_o2
		trigger = { var:votc_dlg_nopt >= 2 }
		votc_execute_staged = { slot = 2 }
		votc_talk_reply = { choice = 2 }
	}
	option = {
		name = votc_dlg_o3
		trigger = { var:votc_dlg_nopt >= 3 }
		votc_execute_staged = { slot = 3 }
		votc_talk_reply = { choice = 3 }
	}
	option = {
		name = votc_dlg_o4
		trigger = { var:votc_dlg_nopt >= 4 }
		votc_execute_staged = { slot = 4 }
		votc_talk_reply = { choice = 4 }
	}
	option = {
		name = votc_dlg_leave
		fallback = yes
		votc_stage_clear = yes
		votc_end_talk = yes
	}
}
